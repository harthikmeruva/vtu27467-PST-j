import java.util.HashSet;

public class Solution {
    public boolean containsDuplicate(int[] nums) {
        // Create a HashSet to keep track of elements we've seen
        HashSet<Integer> seen = new HashSet<>();
        
        for (int num : nums) {
            // If the set already contains the number, a duplicate exists
            if (seen.contains(num)) {
                return true;
            }
            // Otherwise, add the number to the set
            seen.add(num);
        }
        
        // No duplicates found
        return false;
    }

    // Example test runner
    public static void main(String[] args) {
        Solution sol = new Solution();
        
        int[] nums1 = {1, 2, 3, 1};
        int[] nums2 = {1, 2, 3, 4};
        
        System.out.println(sol.containsDuplicate(nums1)); // Output: true
        System.out.println(sol.containsDuplicate(nums2)); // Output: false
    }
}
