import java.util.*;

class Student {
    private int id;
    private String name;
    private double cgpa;

    public Student(int id, String name, double cgpa) {
        this.id = id;
        this.name = name;
        this.cgpa = cgpa;
    }

    public int getID() {
        return id;
    }

    public String getName() {
        return name;
    }

    public double getCGPA() {
        return cgpa;
    }
}

class Priorities {

    public List<Student> getStudents(List<String> events) {

        // Priority rules:
        // 1. Higher CGPA first
        // 2. If CGPA is same, name in ascending alphabetical order
        // 3. If name is same, smaller ID first
        PriorityQueue<Student> pq = new PriorityQueue<>(
            new Comparator<Student>() {
                @Override
                public int compare(Student s1, Student s2) {

                    // Higher CGPA gets higher priority
                    int cgpaCompare = Double.compare(
                        s2.getCGPA(), s1.getCGPA()
                    );

                    if (cgpaCompare != 0) {
                        return cgpaCompare;
                    }

                    // Alphabetical order by name
                    int nameCompare = s1.getName().compareTo(s2.getName());

                    if (nameCompare != 0) {
                        return nameCompare;
                    }

                    // Smaller ID first
                    return Integer.compare(s1.getID(), s2.getID());
                }
            }
        );

        // Process all events
        for (String event : events) {

            String[] parts = event.split(" ");

            if (parts[0].equals("ENTER")) {
                String name = parts[1];
                double cgpa = Double.parseDouble(parts[2]);
                int id = Integer.parseInt(parts[3]);

                pq.add(new Student(id, name, cgpa));

            } else if (parts[0].equals("SERVED")) {

                if (!pq.isEmpty()) {
                    pq.poll();
                }
            }
        }

        // Return remaining students in priority order
        List<Student> result = new ArrayList<>();

        while (!pq.isEmpty()) {
            result.add(pq.poll());
        }

        return result;
    }
}
